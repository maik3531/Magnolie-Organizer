 string Choose(JsonObject book,string mode)=>mode=="original"?book.ToJsonString():mode=="runtime"?ReminderScheduler.SelectRuntimeData(book.ToJsonString()):ReminderScheduler.SelectBackgroundData(book.ToJsonString())!;

 int Deliver(string json,DateTime time,string state){var notices=new List<ReminderNotice>();using var scheduler=new ReminderScheduler(state,notices.Add);scheduler.UpdateData(json);scheduler.RunOnce(time);if(scheduler.LastError is not null)throw new Exception(scheduler.LastError);return notices.Count;}

  var book=Book();var json=Choose(book,mode);var state=Path.Combine(output,"dedup-"+mode+".json");
  foreach(var old in Directory.GetFiles(output,"dedup-"+mode+".json*"))File.Delete(old);
  var keys=Keys(json,now);Check(keys.Count==1,mode+": simultaneous absolute/default due count "+keys.Count);
  var first=Deliver(json,now,state);var restart=Deliver(json,now.AddSeconds(1),state);
  Check(first==1&&restart==0,mode+": actual initial/restart notices "+first+","+restart);
  var distinct=Keys(Choose(Book(seconds:1),mode),now.AddSeconds(1));Check(distinct.Count==2,mode+": distinct seconds were collapsed");

  var legacyBook=book.DeepClone().AsObject();legacyBook["aufgaben"]![0]!["icsRoundtrip"]=new JsonArray("DTSTART:20261127T103000Z","DUE:20261127T103000Z");
  var removed=Deliver(Choose(legacyBook,mode),now.AddSeconds(2),state);Check(removed==0,mode+": removing absolute alarm replays due notification");
  var legacyState=Path.Combine(output,"dedup-existing-"+mode+".json");
  foreach(var old in Directory.GetFiles(output,"dedup-existing-"+mode+".json*"))File.Delete(old);
  var legacyFirst=Deliver(Choose(legacyBook,mode),now,legacyState);var resumed=Deliver(json,now.AddSeconds(1),legacyState);
  Check(legacyFirst==1&&resumed==0,mode+": existing default key replayed "+legacyFirst+","+resumed);
  Console.WriteLine(JsonSerializer.Serialize(new{mode,keys,first,restart,distinctSecondCount=distinct.Count,differentTaskCount=other.Count,removedAbsoluteReplay=removed,legacyFirst,legacyRestart=resumed}));
 }
